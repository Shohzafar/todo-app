CRUD — dasturlash va ma'lumotlar bazasi bilan ishlashda eng ko'p qo'llaniladigan 4 ta asosiy amalning bosh harflaridan olingan qisqartma nomdir.
Har qanday veb-sayt yoki mobil ilova (masalan: Telegram, Instagram, E-Commerce do'konlar, Todo ilovalari) negizida aynan shu 4 ta amal yotadi.

C - CREATE POST - Yangi ma'lumot yaratish, Yangi rasm yoki post joylash.
R - Read - Ma'lumotni o'qish / ko'rish, Lenta (Feed)dagi postlarni ko'rish
U - Update - Ma'lumotni tahrirlash / o'zgartirish, Post matnini (caption) tahrirlash
D - Delete - Ma'lumotni o'chirib tashlash, Postni o'chirib yuborish

def new_todo_page(request):
    if request.method == 'POST':
        title = request.POST.get("title")
        description = request.POST.get("description")

        Todo.objects.create(
            owner=request.user,
            title=title,
            description=description,
        )
        return redirect('/')

    return render(request, "new_todo.html")

from django.shortcuts import render, redirect, get_object_or_404
from .models import Todo

def edit_todo(request, todo_id):
    # 1. Tahrirlanadigan obyektni bazadan topamiz (ID bo'yicha)
    todo = get_object_or_404(Todo, id=todo_id)

    # 2. Agar foydalanuvchi "Saqlash" tugmasini bossa (POST)
    if request.method == 'POST':
        # Formadan kelgan yangi qiymatlarni olamiz
        todo.title = request.POST.get('title')
        todo.description = request.POST.get('description')
        
        # 3. AYNAN SHU QATOR UPDATE AMALINI BAJARADI
        todo.save()

        # O'zgartirib bo'lgach, bosh sahifaga qaytaramiz
        return redirect('home')

    # 4. Agar sahifaga shunchaki kirilgan bo'lsa (GET), eskilarini ko'rsatamiz
    return render(request, 'edit_todo.html', {'todo': todo})

# app_main/views.py
from django.shortcuts import render, redirect, get_object_or_404
from .models import Todo

# 1. Bosh sahifa (Read)
def home_page(request):
    todos = Todo.objects.all()
    return render(request, 'app_main/home.html', {'todos': todos})

# 2. Yangi Todo yaratish sahifasi va amali (Create)
def new_todo_page(request):
    if request.method == 'POST':
        title = request.POST.get('title')
        description = request.POST.get('description')
        
        Todo.objects.create(title=title, description=description)
        return redirect('home')
        
    return render(request, 'app_main/new_todo.html')

# 3. Todo tahrirlash sahifasi va amali (Update)
def edit_todo_page(request, todo_id):
    todo = get_object_or_404(Todo, id=todo_id)
    
    if request.method == 'POST':
        todo.title = request.POST.get('title')
        todo.description = request.POST.get('description')
        todo.save()  # UPDATE amali
        return redirect('home')
        
    return render(request, 'app_main/edit_todo.html', {'todo': todo})

# 4. Todo o'chirish amali (Delete)
def delete_todo(request, todo_id):
    todo = get_object_or_404(Todo, id=todo_id)
    if request.method == 'POST':
        todo.delete()  # DELETE amali
    return redirect('home')

# app_main/urls.py
from django.urls import path
from . import views  # Shu papkadagi views.py ni import qilamiz

urlpatterns = [
    path('', views.home_page, name='home'),
    path('new-todo/', views.new_todo_page, name='new_todo'),
    path('edit-todo//', views.edit_todo_page, name='edit_todo'),
    path('delete-todo//', views.delete_todo, name='delete_todo'),
]
